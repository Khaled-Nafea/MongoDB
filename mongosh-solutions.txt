// Q1: Create an explicit collection named "books" with a
// validation rule to ensure that each document has a non-empty
// "title" field.

db.createCollection("books", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: {
          bsonType: "string",
          minLength: 1,
          description: "title must be a non-empty string and is required"
        }
      }
    }
  },
  validationLevel: "strict",
  validationAction: "error"
});
// =============================================================
// Q2: Create an implicit collection by inserting data directly
// into a new collection named "authors".
 
db.authors.insertOne({
  name: "Author1",
  nationality: "British"
});
// =============================================================
// Q3: Create a capped collection named “logs” with a size limit of 1MB.

db.createCollection("logs", { capped: true, size: 1048576 });
// =============================================================
// Q4: Create an index on the books collection for the title field.

db.books.createIndex({ title: 1 });
// =============================================================
// Q5: Insert one document into the books collection.

db.books.insertOne({
  title: "Book1",
  author: "Ali",
  year: 1937,
  genres: ["fantasy", "Adventure"]
});
// =============================================================
// Q6: Insert multiple documents into books collection
db.books.insertMany([
  {
    title: "Future",
    author: "Ahmed",
    year: 2020,
    genres: ["Sinence Fiction"]
  },
  {
    title: "Pyramids",
    author: "Sara",
    year: 2021,
    genres: ["History"]
  },
  {
    title: "Cosmos",
    author: "Mona",
    year: 2022,

  }

// =============================================================
// Q7: Insert a new log into the logs collection.

db.logs.insertOne({
  book_id: ObjectId("6ab6f8218bd2d9ccc823ccaa"),
  action: "borrowed"
});
// =============================================================
// Q8: Update the book with title "Future" change the year to be 2022.

db.books.updateOne(
  { title: "Future" },
  { $set: { year: 2022 } }
);
// =============================================================
// Q9: Find a Book with title containing "Brave"

db.books.findOne({
  title: { $regex: "Brave", $options: "i" }
});
// =============================================================
// Q10: Find all books published between 1990 and 2010

db.books.find({
  year: { $gte: 1990, $lte: 2010 }
});
// =============================================================
// Q11: Find books where the genre includes "Science Fiction"

db.books.find({
  genres: "Science Fiction"
});
// =============================================================
// Q12: Skip the first two books, limit to next 3, sorted by year descending

db.books.find()
  .sort({ year: -1 })
  .skip(2)
  .limit(3);
// =============================================================
// Q13: Find books where the year field is stored as an integer

db.books.find({
  year: { $type: "int" }
});
// =============================================================
// Q14: Find books where genres does not include "Horror" or "Science Fiction"

db.books.find({
  genres: { $nin: ["Horror", "Science Fiction"] }
});
// =============================================================
// Q15: Delete all books published before 2000

db.books.deleteMany({
  year: { $lt: 2000 }
});
// =============================================================
// Q16: Aggregate: Filter books published after 2000 and sort by year descending

db.books.aggregate([
  {
    $match: {
      year: { $gt: 2000 }
    }
  },
  {
    $sort: {
      year: -1
    }
  }
]);
// =============================================================
// Q17: Aggregate: Books published after 2000 showing only title, author, and year

db.books.aggregate([
  {
    $match: {
      year: { $gt: 2000 }
    }
  },
  {
    $project: {
      _id: 0,
      title: 1,
      author: 1,
      year: 1
    }
  }
]);
// =============================================================
// Q18: Aggregate: Break genres array into separate documents showing title and genres only

db.books.aggregate([
  {
    $unwind: "$genres"
  },
  {
    $project: {
      _id: 0,
      title: 1,
      genres: 1
    }
  }
]);
